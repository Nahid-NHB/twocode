# Docker Sandbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Isolate `bash` tool execution inside a Docker container so AI-generated shell commands can't reach outside the project directory, steal credentials, or phone home.

**Architecture:** New `@twocode/sandbox` package exposes `runInSandbox(command, cwd)` and `isDockerAvailable()`. The CLI's `executeLocalTool` bash case is replaced with a call to `runInSandbox`. A startup component checks Docker availability and shows a toast if it's missing.

**Tech Stack:** Bun, Docker CLI, Alpine Linux (container image), `bun:test`

## Global Constraints

- Bun runtime only — no Node.js `child_process`, use `Bun.spawn`
- No new npm dependencies — Docker CLI is already on the host
- TypeScript strict mode (`tsconfig.base.json` is extended by each package)
- All test files under `src/__tests__/` in the package under test
- Tests require Docker running — document this, don't mock Docker
- Container image tag: `twocode-sandbox`
- Timeout: 30 000 ms — matches existing `DEFAULT_TIMEOUT` in `local-tools.ts`
- Output truncation limit: 20 000 chars — matches existing `MAX_OUTPUT`

```typescript
import { resolve } from "path";

const IMAGE_TAG = "twocode-sandbox";
const PACKAGE_DIR = resolve(import.meta.dir, "..");

let buildPromise: Promise<void> | null = null;

export async function isDockerAvailable(): Promise<boolean> {
  const proc = Bun.spawn(["docker", "info"], {
    stdout: "ignore",
    stderr: "ignore",
  });
  await proc.exited;
  return proc.exitCode === 0;
}

export async function ensureImage(): Promise<void> {
  if (buildPromise) return buildPromise;
  buildPromise = _buildIfNeeded();
  return buildPromise;
}

async function _buildIfNeeded(): Promise<void> {
  const check = Bun.spawn(["docker", "images", IMAGE_TAG, "-q"], {
    stdout: "pipe",
    stderr: "ignore",
  });
  const output = await new Response(check.stdout).text();
  await check.exited;
  if (output.trim()) return;

  const build = Bun.spawn(
    ["docker", "buildx", "build", "-t", IMAGE_TAG, "."],
    { cwd: PACKAGE_DIR, stdout: "ignore", stderr: "pipe" },
  );
  const stderr = await new Response(build.stderr).text();
  await build.exited;
  if (build.exitCode !== 0) {
    throw new Error(`Failed to build sandbox image:\n${stderr}`);
  }
}
```

- [ ] **Step 3: Implement index.ts**

```typescript
import { randomUUID } from "crypto";
import { ensureImage, isDockerAvailable } from "./image";

export { isDockerAvailable } from "./image";

export interface SandboxResult {
  stdout: string;
  stderr: string;
  exitCode: number;
}

const MAX_OUTPUT = 20_000;
const TIMEOUT_MS = 30_000;

function truncate(value: string, limit: number): string {
  return value.length > limit
    ? `${value.slice(0, limit)}\n... (truncated, ${value.length} total chars)`
    : value;
}

export async function runInSandbox(
  command: string,
  cwd: string,
): Promise<SandboxResult> {
  const available = await isDockerAvailable();
  if (!available) {
    throw new Error("bash tool requires Docker. Start Docker and try again.");
  }

  await ensureImage();

  const name = `twocode-${randomUUID().slice(0, 8)}`;

  const proc = Bun.spawn(
    [
      "docker", "run",
      "--rm",
      "--name", name,
      "--network=none",
      "--no-new-privileges",
      "--cap-drop=ALL",
      "--read-only",
      "--tmpfs", "/tmp",
      "-m", "512m",
      "--cpus=1",
      "-v", `${cwd}:${cwd}:rw`,
      "--workdir", cwd,
      "twocode-sandbox",
      "bash", "-c", command,
    ],
    { stdout: "pipe", stderr: "pipe" },
  );

  let timedOut = false;
  const killTimer = setTimeout(() => {
    timedOut = true;
    Bun.spawn(["docker", "kill", name], { stdout: "ignore", stderr: "ignore" });
  }, TIMEOUT_MS);

  const [stdout, stderr] = await Promise.all([
    new Response(proc.stdout).text(),
    new Response(proc.stderr).text(),
  ]);
  const exitCode = await proc.exited;
  clearTimeout(killTimer);

  if (timedOut) {
    return {
      stdout: truncate(stdout, MAX_OUTPUT),
      stderr: "timed out after 30s",
      exitCode: 124,
    };
  }

  return {
    stdout: truncate(stdout, MAX_OUTPUT),
    stderr: truncate(stderr, MAX_OUTPUT),
    exitCode: exitCode ?? 1,
  };
}
```

```typescript
}
```
